import java.util.concurrent.ConcurrentHashMap;

/**
 * EnterpriseDemoApplication.java
 *
 * Large single-file Java application example.
 */
public class EnterpriseDemoApplication {

    public static void main(String[] args) {
        UserRepository userRepository = new InMemoryUserRepository();
        AuditService auditService = new AuditService();
        NotificationService notificationService = new EmailNotificationService();

        UserService userService =
                new UserService(userRepository, auditService, notificationService);

        userService.createUser(
                new CreateUserRequest(
                        "john.doe@company.com",
                        "John",
                        "Doe",
                        Role.ADMIN));

        userService.createUser(
                new CreateUserRequest(
                        "alice.smith@company.com",
                        "Alice",
                        "Smith",
                        Role.USER));

        userService.createUser(
                new CreateUserRequest(
                        "bob.wilson@company.com",
                        "Bob",
                        "Wilson",
                        Role.MANAGER));

        System.out.println("\n=== ALL USERS ===");
        userService.getAllUsers().forEach(System.out::println);

        System.out.println("\n=== ACTIVE USERS ===");
        userService.getUsersByStatus(UserStatus.ACTIVE)
                .forEach(System.out::println);

        System.out.println("\n=== SEARCH RESULT ===");
        userService.searchUsers("alice")
                .forEach(System.out::println);

        System.out.println("\n=== AUDIT LOGS ===");
        auditService.getAuditEntries()
                .forEach(System.out::println);
    }
}

class User {

    private UUID id;
    private String email;
    private String firstName;
    private String lastName;
    private UserStatus status;
    private Role role;
    private LocalDateTime createdAt;

    public User() {
    }

    public User(
            UUID id,
            String email,
            String firstName,
            String lastName,
            UserStatus status,
            Role role,
            LocalDateTime createdAt) {

        this.id = id;
        this.email = email;
        this.firstName = firstName;
        this.lastName = lastName;
        this.status = status;
        this.role = role;
        this.createdAt = createdAt;
    }

    public UUID getId() {
        return id;
    }

    public void setId(UUID id) {
        this.id = id;
    }

    public String getEmail() {
        return email;
    }

    public void setEmail(String email) {
        ValidationUtil.requireNonEmpty(email, "Email");
        this.email = email;
    }

    public String getFirstName() {
        return firstName;
    }

    public void setFirstName(String firstName) {
        ValidationUtil.requireNonEmpty(firstName, "First name");
        this.firstName = firstName;
    }

    public String getLastName() {
        return lastName;
    }

    public void setLastName(String lastName) {
        ValidationUtil.requireNonEmpty(lastName, "Last name");
        this.lastName = lastName;
    }

    public UserStatus getStatus() {
        return status;
    }

    public void setStatus(UserStatus status) {
        this.status = status;
    }

    public Role getRole() {
        return role;
    }

    public void setRole(Role role) {
        this.role = role;
    }

    public LocalDateTime getCreatedAt() {
        return createdAt;
    }

    public void setCreatedAt(LocalDateTime createdAt) {
        this.createdAt = createdAt;
    }

    public String getFullName() {
        return firstName + " " + lastName;
    }

    @Override
    public String toString() {
        return "User{" +
                "id=" + id +
                ", email='" + email + '\'' +
                ", fullName='" + getFullName() + '\'' +
                ", status=" + status +
                ", role=" + role +
                ", createdAt=" + createdAt +
                '}';
    }
}

/* ============================
   REPOSITORIES
   ============================ */

interface UserRepository {

    User save(User user);

    Optional<User> findById(UUID id);

    Optional<User> findByEmail(String email);

    List<User> findAll();

    void delete(UUID id);
}

class InMemoryUserRepository implements UserRepository {

    private final Map<UUID, User> storage =
            new ConcurrentHashMap<>();

    @Override
    public User save(User user) {
        storage.put(user.getId(), user);
        return user;
    }

    @Override
    public Optional<User> findById(UUID id) {
        return Optional.ofNullable(storage.get(id));
    }

    @Override
    public Optional<User> findByEmail(String email) {

        return storage.values()
                .stream()
                .filter(u -> u.getEmail()
                        .equalsIgnoreCase(email))
                .findFirst();
    }

    @Override
    public List<User> findAll() {
        return new ArrayList<>(storage.values());
    }

    @Override
    public void delete(UUID id) {
        storage.remove(id);
    }
}

/* ============================
   SERVICES
   ============================ */

class UserService {

    private final UserRepository userRepository;
    private final AuditService auditService;
    private final NotificationService notificationService;

    public UserService(
            UserRepository userRepository,
            AuditService auditService,
            NotificationService notificationService) {

        this.userRepository = userRepository;
        this.auditService = auditService;
        this.notificationService = notificationService;
    }

    public User createUser(
            CreateUserRequest request) {

        ValidationUtil.requireValidEmail(
                request.getEmail());

        userRepository.findByEmail(
                request.getEmail())
                .ifPresent(u -> {
                    throw new DuplicateUserException(
                            "User already exists");
                });

        User user = new User();

        user.setId(UUID.randomUUID());
        user.setEmail(request.getEmail());
        user.setFirstName(request.getFirstName());
        user.setLastName(request.getLastName());
        user.setRole(request.getRole());
        user.setStatus(UserStatus.ACTIVE);
        user.setCreatedAt(LocalDateTime.now());

        userRepository.save(user);

        auditService.log(
                AuditAction.USER_CREATED,
                user.getEmail());

        notificationService.sendNotification(
                user.getEmail(),
                "Welcome " + user.getFullName());

        return user;
    }

    public List<User> getAllUsers() {
        return userRepository.findAll();
    }

    public List<User> getUsersByStatus(
            UserStatus status) {

        return userRepository.findAll()
                .stream()
                .filter(u -> u.getStatus() == status)
                .collect(Collectors.toList());
    }

    public List<User> searchUsers(String term) {

        String normalized =
                term.toLowerCase();

        return userRepository.findAll()
                .stream()
                .filter(u ->
                        u.getEmail().toLowerCase().contains(normalized)
                                || u.getFullName()
                                .toLowerCase()
                                .contains(normalized))
                .collect(Collectors.toList());
    }
}

class AuditService {

    private final List<AuditEntry> entries =
            new ArrayList<>();

    public void log(
            AuditAction action,
            String target) {

        AuditEntry entry =
                new AuditEntry(
                        UUID.randomUUID(),
                        action,
                        target,
                        LocalDateTime.now());

        entries.add(entry);
    }

    public List<AuditEntry> getAuditEntries() {
        return Collections.unmodifiableList(entries);
    }
}

interface NotificationService {
    void sendNotification(
            String recipient,
            String message);
}
